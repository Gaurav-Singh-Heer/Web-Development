Sure! Let's break it down clearly:

### The issue:
In **React**, when you use `useState()`, you should **never directly modify the state array/object**, because React depends on immutability to detect changes and re-render components correctly.

### Your `state` is an array:

```js
const [state, setState] = useState(Array(9).fill(null));
```

So, `state` looks like this initially:

```js
[null, null, null, null, null, null, null, null, null]
```

---

### Why `copyState`?

When the user clicks a square, we need to "mark" it with `"X"` or `"O"`. But instead of doing this:
```js
state[index] = "X"; // ❌ BAD (Direct mutation)
```

We do:
```js
const copyState = [...state]; // ✅ GOOD (Create a shallow copy)
copyState[index] = isXTurn ? "X" : "O"; // Modify the copy
setState(copyState); // Update the state
```

### Explanation of `const copyState = [...state];`

`[...state]` is the **spread operator** which creates a shallow copy of the `state` array. It’s like saying:

```js
const copyState = Array.from(state);
```

This ensures you don’t mutate the original `state` array directly. React will now recognize that a new array reference has been created and trigger a re-render.

---

### Quick Analogy:

Think of `state` like an "official record book". You don't scribble directly on the official book; instead, you make a **photocopy**, mark changes on the copy, and then replace the old book with the updated version!

---

### Why React cares:
React uses **shallow comparison** (`===`) to check if your state has changed. If you mutate `state` directly, the array reference doesn’t change, and React might skip re-rendering.

import React, {useState} from "react";
import Square from "./Square";

const Board = () => {
    const [state, setState] = useState(Array(9).fill(null))
    const [isXTurn, setIsXTurn] = useState(true);

    const handleClick = (index) => {
        if (state[index]) return; // Prevent overwriting a square

        console.log("Clicked on Index", index);
        
        const copyState = [...state];
        copyState[index] = isXTurn ? "X" : "O";
        setState(copyState);
        setIsXTurn(!isXTurn);
    };

    console.log("State", state);

    return (
        <div className="board-container">
            <div className="board-row">
                <Square onClick={() => handleClick(0)} value={state[0]}/>
                <Square onClick={() => handleClick(1)} value={state[1]}/>
                <Square onClick={() => handleClick(2)} value={state[2]}/>
            </div>
            <div className="board-row">
                <Square onClick={() => handleClick(3)} value={state[3]}/>
                <Square onClick={() => handleClick(4)} value={state[4]}/>
                <Square onClick={() => handleClick(5)} value={state[5]}/>
            </div>
            <div className="board-row">
                <Square onClick={() => handleClick(6)} value={state[6]}/>
                <Square onClick={() => handleClick(7)} value={state[7]}/>
                <Square onClick={() => handleClick(8)} value={state[8]}/>
            </div>
        </div>
    )
}

import React, {useState} from "react";
import Square from "./Square";

const Board = () => {
    const [state, setState] = useState(Array(9).fill(null))
    const [isXTurn, setIsXTurn] = useState(true);

    const checkWinner = () =>{
        const winnerLogic = [
            [0, 1, 2],
            [3, 4, 5],
            [6, 7, 8],
            [0, 3, 6],
            [1, 4, 7],
            [0, 4, 8],
            [2, 4, 6],
        ]

        for (let logic of winnerLogic) {
            const [a, b, c] = logic;

            if(state[a] === state[b] && state[a] === state[c]){
                return state[a];
            }
        }
        return false;
    }

    const isWinner = checkWinner();
    const isDraw = !isWinner && state.every((square) => square !== null);

    const handleClick = (index) => {
        if (state[index]) return; // Prevent overwriting a square

        console.log("Clicked on Index", index);
        
        const copyState = [...state];
        copyState[index] = isXTurn ? "X" : "O";
        setState(copyState);
        setIsXTurn(!isXTurn);
    };

    console.log("State", state);

    const resetGame = () => {
        setState(Array(9).fill(null));
        setIsXTurn(true);
    };

    return (
        <div className="board-container">
            {isWinner ? (
                <div className="result">
                    <h2>🎉 {isWinner} won the game!</h2>
                    <button className="restart-btn" onClick={resetGame}>
                        Play Again
                    </button>
                </div>
            ) : isDraw ? (
                <div className="result">
                    <h2>🤝 It's a Draw!</h2>
                    <button className="restart-btn" onClick={resetGame}>
                        Restart Game
                    </button>
                </div>
            ) : (
            <>
              <h3>Next Turn: {isXTurn ? "X" : "O"}</h3>
              <div className="board-row">
                <Square onClick={() => handleClick(0)} value={state[0]} />
                <Square onClick={() => handleClick(1)} value={state[1]} />
                <Square onClick={() => handleClick(2)} value={state[2]} />
              </div>
              <div className="board-row">
                <Square onClick={() => handleClick(3)} value={state[3]} />
                <Square onClick={() => handleClick(4)} value={state[4]} />
                <Square onClick={() => handleClick(5)} value={state[5]} />
              </div>
              <div className="board-row">
                <Square onClick={() => handleClick(6)} value={state[6]} />
                <Square onClick={() => handleClick(7)} value={state[7]} />
                <Square onClick={() => handleClick(8)} value={state[8]} />
              </div>
            </>
            )}
        </div>
    );
}

export default Board;
