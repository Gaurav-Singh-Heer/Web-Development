const fs=require("fs");

-------------------------------------------------------------------------------------------------
1) Write Synchronous

// Sync....
fs.writeFileSync("./3_1_test.txt","Hey There")
fs.writeFileSync("./3_1_test.txt","Hello world") // this will overwrite "Hey There" in file

// Used to Write data in File

-------------------------------------------------------------------------------------------------
2) Write Asynchronous

// Async....
fs.writeFile("./3_2_test.txt","Hey There",(err)=>{}) 
fs.writeFile("./3_2_test.txt","Hello world Async",(err)=>{}) // this will overwrite "Hey There" in file

// (err)=>{} is to handle errors

-------------------------------------------------------------------------------------------------
3) Read Synchronous

const result = fs.readFileSync("./3_3_Contact.txt","utf-8");
// ReadFileSync-> Gives Result in store in variable

// Used to Read data from File

-------------------------------------------------------------------------------------------------
4) Read Asynchronous

fs.readFile("./3_3_Contact.txt","utf-8",(err,result)=>{
    if(err){
        console.log("Error",err);
    }
    else{
        console.log(result)
    }
})

-------------------------------------------------------------------------------------------------

Summary of 1,2,3,4->

Whenever we do Synchronous task it returns the result and if error is there we can use 
try catch to handle it 

Whenever we do Asynchronous task it does not returns the result rather is always expect a 
callback (like err)

5) Append File Sync
