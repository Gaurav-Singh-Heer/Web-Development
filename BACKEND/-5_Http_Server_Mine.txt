const myServer=http.createServer((req,res)=>{});

This arrow function is responsible to process our incoming request.
This callback function has 2 arguments/parameters:
1) req-> For request
2) res-> For response 

Whenever there is incoming call/request at our server it runs this callback function:-

req, First argument will have all information about the request (user,userIP,userRequest)
res, To send response

Created a server to log data Whenever opened

const http=require("http")
const fs=require("fs")

const myServer=http.createServer((req,res)=>{
    const log=`${Date.now()}: New Request Received\n`;
    fs.appendFile('5_6_log.txt',log,(err,data)=>{
        res.end("Hello From Server");
    }); // not using synchronous as if all thread in work user has to wait
})

myServer.listen(8000,()=>console.log("Server Started!"));

/*
    appendFile automatically creates file when server is opened

    By default, 2 entries are logged each time
*/

const http=require("http")
const fs=require("fs")

const myServer=http.createServer((req,res)=>{
    const log=`${Date.now()}:${req.url} New Request Received\n`;
    fs.appendFile('5_8_log.txt',log,(err,data)=>{
        switch(req.url){
            case '/':
                res.end("HomePage");
            break
            case '/about':
                res.end("I am Gaurav Singh Heer");
            break
            default:
                res.end("404 Not found");
            break
        }
    }); // not using synchronous as if all thread in work user has to wait
})

myServer.listen(8000,()=>console.log("Server Started!"));
