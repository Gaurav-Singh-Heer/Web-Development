---> routes/28_blog.js, modify the upload configuration to use multer-storage-cloudinary. so made routes/30_blog.js

|                           const { Router } = require("express");
|                           const multer = require("multer");
|                           const { CloudinaryStorage } = require("multer-storage-cloudinary");
|                           const cloudinary = require("../config/cloudinary"); // Import cloudinary config
|                           
|                           const Blog = require("../models/27_blogs");
|                           const Comment = require("../models/28_comment");
|                           
|                           const router = Router();
|                           
|                           // Configure Cloudinary Storage
|                           
|                           const storage = new CloudinaryStorage({
|                               cloudinary: cloudinary,
|                               params: {
|                                   folder: "blog_images", // Folder in Cloudinary
|                                   format: async (req, file) => "jpg", // Force file format to jpg
|                                   public_id: (req, file) => `${Date.now()}-${file.originalname.split(".")[0]}`, // Generate public ID
|                               },
|                           });
|                           
|                           const upload = multer({ storage });
|                           
|                           router.post("/", upload.single("coverImage"), async (req, res) => {
|                               try {
|                                   const { title, body } = req.body;
|                                   const blog = await Blog.create({
|                                       title,
|                                       body,
|                                       createdBy: req.user._id,
|                                       coverImageURL: req.file.path, // URL from Cloudinary
|                                       coverImagePublicId: req.file.filename, // Cloudinary public ID
|                                   });
|                                   return res.redirect(`/blog/${blog._id}`);
|                               } catch (error) {
|                                   console.error(error);
|                                   return res.status(500).send("Error uploading blog");
|                               }
|                           });
|                           
|                           module.exports = router;
